package com.Laimis.Java_Game;

import com.badlogic.gdx.ApplicationAdapter;
import com.badlogic.gdx.Gdx;
import com.badlogic.gdx.Input;
import com.badlogic.gdx.graphics.GL20;
import com.badlogic.gdx.graphics.OrthographicCamera;
import com.badlogic.gdx.graphics.Texture;
import com.badlogic.gdx.graphics.g2d.Animation;
import com.badlogic.gdx.graphics.g2d.SpriteBatch;
import com.badlogic.gdx.graphics.g2d.TextureRegion;
import com.badlogic.gdx.maps.tiled.TiledMap;
import com.badlogic.gdx.maps.tiled.TiledMapRenderer;
import com.badlogic.gdx.maps.tiled.TmxMapLoader;
import com.badlogic.gdx.maps.tiled.renderers.OrthogonalTiledMapRenderer;
import com.badlogic.gdx.math.Rectangle;
import com.badlogic.gdx.math.Vector2;
import com.badlogic.gdx.utils.Array;

public class Main extends ApplicationAdapter {
    SpriteBatch batch;
    Texture playerTexture, bulletTexture, enemyTexture;
    Animation<TextureRegion> playerAnimation;
    float stateTime;
    Vector2 playerPosition;
    Array<Vector2> bullets;
    Array<Rectangle> enemies;
    OrthographicCamera camera;
    float playerSpeed = 200;
    float bulletSpeed = 400;
    float playerScale = 0.2f*0.2f;

    float gravity = -600;
    float jumpForce = 300;
    float velocityY = 0;
    boolean isOnGround = true;

    TiledMap tiledMap;
    TiledMapRenderer tiledMapRenderer;

    @Override
    public void create() {

        batch = new SpriteBatch();
        playerTexture = new Texture("C:\\Users\\laimi\\Desktop\\VU_PASKAITOS\\Semester_2\\Java\\Java_Game\\assets\\player.png");
        bulletTexture = new Texture("C:\\Users\\laimi\\Desktop\\VU_PASKAITOS\\Semester_2\\Java\\Java_Game\\assets\\bullet.png");
        enemyTexture = new Texture("C:\\Users\\laimi\\Desktop\\VU_PASKAITOS\\Semester_2\\Java\\Java_Game\\assets\\enemy.png");
        playerPosition = new Vector2(100, 300);
        bullets = new Array<>();
        enemies = new Array<>();

        // Load player animation
        TextureRegion[][] tmp = TextureRegion.split(playerTexture, playerTexture.getWidth() / 4, playerTexture.getHeight());
        TextureRegion[] playerFrames = new TextureRegion[4];
        for (int i = 0; i < 4; i++) {
            playerFrames[i] = tmp[0][i];
        }
        playerAnimation = new Animation<>(0.1f, playerFrames);
        stateTime = 0f;

        // Initialize camera
        camera = new OrthographicCamera(Gdx.graphics.getWidth(), Gdx.graphics.getHeight());
        camera.position.set(playerPosition.x, playerPosition.y, 0);

        // Load Tiled map
        tiledMap = new TmxMapLoader().load("level1.tmx");
        tiledMapRenderer = new OrthogonalTiledMapRenderer(tiledMap);

        // Add some enemies
        enemies.add(new Rectangle(400, 100, 32, 32));
        enemies.add(new Rectangle(600, 100, 32, 32));


    }

    @Override
    public void render() {
        update(Gdx.graphics.getDeltaTime());

        Gdx.gl.glClear(GL20.GL_COLOR_BUFFER_BIT);

        camera.update();
        tiledMapRenderer.setView(camera);
        tiledMapRenderer.render();

        batch.setProjectionMatrix(camera.combined);
        batch.begin();

        // Draw player with animation
        stateTime += Gdx.graphics.getDeltaTime();
        TextureRegion currentFrame = playerAnimation.getKeyFrame(stateTime, true);
        batch.draw(currentFrame, playerPosition.x, playerPosition.y, playerTexture.getWidth() / 4 * playerScale, playerTexture.getHeight() * playerScale);


        // Draw bullets
        for (Vector2 bullet : bullets) {
            batch.draw(bulletTexture, bullet.x, bullet.y);
        }

        // Draw enemies
        for (Rectangle enemy : enemies) {
            batch.draw(enemyTexture, enemy.x, enemy.y);
        }

        batch.end();
    }

    public void update(float deltaTime) {
        handleInput(deltaTime);
        applyGravity(deltaTime);
        updateBullets(deltaTime);
        checkCollisions();

        updateCamera();
    }

    private void handleInput(float deltaTime) {
        if (Gdx.input.isKeyPressed(Input.Keys.LEFT)) {
            playerPosition.x -= playerSpeed * deltaTime;
        }
        if (Gdx.input.isKeyPressed(Input.Keys.RIGHT)) {
            playerPosition.x += playerSpeed * deltaTime;
        }
        if (Gdx.input.isKeyJustPressed(Input.Keys.SPACE)) {
            bullets.add(new Vector2(playerPosition.x + 40, playerPosition.y + 20));
        }
        if (Gdx.input.isKeyJustPressed(Input.Keys.UP) && isOnGround) {
            velocityY = jumpForce;
            isOnGround = false;
        }
    }

    private void applyGravity(float deltaTime) {
        velocityY += gravity * deltaTime;
        playerPosition.y += velocityY * deltaTime;

        // Ground collision
        if (playerPosition.y <= 100) {
            playerPosition.y = 100;
            velocityY = 0;
            isOnGround = true;
        }
    }

    private void updateBullets(float deltaTime) {
        // Move bullets
        for (Vector2 bullet : bullets) {
            bullet.x += bulletSpeed * deltaTime;
        }

        // Remove bullets that go off-screen
        Array<Vector2> bulletsToRemove = new Array<>();
        for (Vector2 bullet : bullets) {
            if (bullet.x > camera.position.x + Gdx.graphics.getWidth() / 2) {
                bulletsToRemove.add(bullet);
            }
        }
        bullets.removeAll(bulletsToRemove, true);
    }

    private void checkCollisions() {
        // Check bullet-enemy collisions
        for (Vector2 bullet : bullets) {
            for (Rectangle enemy : enemies) {
                if (enemy.contains(bullet.x, bullet.y)) {
                    enemies.removeValue(enemy, true);
                    bullets.removeValue(bullet, true);
                    break;
                }
            }
        }
    }

    private void updateCamera() {
        camera.position.set(playerPosition.x + 100, playerPosition.y + 50, 0);
        camera.update();
    }

    @Override
    public void dispose() {
        batch.dispose();
        playerTexture.dispose();
        bulletTexture.dispose();
        enemyTexture.dispose();
        tiledMap.dispose();
    }


}
